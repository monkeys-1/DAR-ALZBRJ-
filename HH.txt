    <script>
        // دالة التبديل بين الشاشات
        function switchTab(viewId, element) {
            // إزالة التفعيل من كل التبويبات
            document.querySelectorAll('.tab-btn').forEach(btn => btn.classList.remove('active'));
            document.querySelectorAll('.view-section').forEach(view => view.classList.remove('active'));
            
            // تفعيل التبويب المختار
            element.classList.add('active');
            document.getElementById(viewId + '-view').classList.add('active');
        }

        // --- الحساب الفوري للباركيه ---
        const pLen = document.getElementById('p-length');
        const pWid = document.getElementById('p-width');
        const pType = document.getElementById('p-type');

        function calculateParquet() {
            let l = parseFloat(pLen.value) || 0;
            let w = parseFloat(pWid.value) || 0;
            let divisor = parseFloat(pType.value);

            let area = l * w;
            document.getElementById('p-area-out').innerText = area.toFixed(2) + ' م²';

            if (area > 0) {
                let rawBoxes = area / divisor; // الناتج الدقيق قبل التقريب
                let fraction = rawBoxes - Math.floor(rawBoxes); // استخراج الجزء العشري فقط (مثل 0.1 أو 0.5)
                let boxes = Math.ceil(rawBoxes); // التقريب للأعلى كخطوة أساسية

                // تطبيق شرط الهدر الذكي:
                // إذا كان الكسر أكبر من 0.2 (مثل 13.5) أو كان العدد بدون كسور (0) -> نضيف كرتون إضافي
                // أما إذا كان الكسر صغيراً (مثل 13.1) -> لا نضيف كرتون إضافي لأن التقريب للأعلى وفر لنا الهدر الكافي
                if (fraction > 0.2 || fraction === 0) {
                    boxes += 1;
                }

                document.getElementById('p-boxes-out').innerText = boxes;
            } else {
                document.getElementById('p-boxes-out').innerText = '0';
            }
        }

        // إضافة أحداث استماع للتحديث التلقائي (مثل AppSheet)
        [pLen, pWid, pType].forEach(el => el.addEventListener('input', calculateParquet));


        // --- الحساب الفوري للجدران ---
        const wWidth = document.getElementById('w-width');
        const wType = document.getElementById('w-type');
        const wInsulation = document.getElementById('w-insulation');

        function calculateWalls() {
            let h = 2.90; // الارتفاع ثابت بناءً على التعديل
            let w = parseFloat(wWidth.value) || 0;
            let isWpc = wType.value === 'wpc';
            let hasInsulation = wInsulation.checked;

            let area = h * w;
            document.getElementById('w-area-out').innerText = area.toFixed(2) + ' م²';

            if (w > 0) {
                // حساب عدد ألواح التكسية (عرض الجدار مقسوماً على عرض اللوح 1.20)
                let boards = Math.ceil(w / 1.20);
                document.getElementById('w-boards-out').innerText = boards;

                // إظهار/إخفاء وحساب العازل (Show_If Logic)
                const insRow = document.getElementById('insulation-row');
                if (hasInsulation) {
                    let insBoards = Math.ceil(w / 1.20); // عرض العازل أيضاً 1.20
                    document.getElementById('w-ins-out').innerText = insBoards;
                    insRow.classList.remove('hidden');
                } else {
                    insRow.classList.add('hidden');
                }

                // إظهار/إخفاء وحساب تكلفة بديل الشيبورد (Show_If Logic)
                const costRow = document.getElementById('cost-row');
                if (isWpc) {
                    let cost = boards * 90;
                    document.getElementById('w-cost-out').innerText = cost + ' ريال';
                    costRow.classList.remove('hidden');
                } else {
                    costRow.classList.add('hidden');
                }

            } else {
                document.getElementById('w-boards-out').innerText = '0';
                document.getElementById('insulation-row').classList.add('hidden');
                document.getElementById('cost-row').classList.add('hidden');
            }
        }

        // إضافة أحداث استماع للتحديث التلقائي
        [wWidth, wType, wInsulation].forEach(el => el.addEventListener('input', calculateWalls));

    </script>
